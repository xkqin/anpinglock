cmake_minimum_required(VERSION 3.20)
project(AnpingLockNative LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(AnpingLock.Agent WIN32 main.cpp agent.rc)
target_compile_definitions(AnpingLock.Agent PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AnpingLock.Agent PRIVATE /W4 /EHsc /permissive- /utf-8 /O1 /Gy /guard:cf)
target_link_options(AnpingLock.Agent PRIVATE /OPT:REF /OPT:ICF /guard:cf)
target_link_libraries(AnpingLock.Agent PRIVATE windowsapp wtsapi32 advapi32 shell32 user32)
add_executable(AnpingLock.NativePolicyTests ../../tests/AnpingLock.NativePolicyTests/main.cpp)
target_include_directories(AnpingLock.NativePolicyTests PRIVATE .)
target_compile_options(AnpingLock.NativePolicyTests PRIVATE /W4 /EHsc /utf-8)
add_executable(AnpingLock.NativeBluetoothProbe ../../tests/AnpingLock.NativeBluetoothProbe/main.cpp)
target_compile_definitions(AnpingLock.NativeBluetoothProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AnpingLock.NativeBluetoothProbe PRIVATE /W4 /EHsc /permissive- /utf-8 /O1 /Gy)
target_link_options(AnpingLock.NativeBluetoothProbe PRIVATE /OPT:REF /OPT:ICF)
target_link_libraries(AnpingLock.NativeBluetoothProbe PRIVATE windowsapp wtsapi32 advapi32 shell32 user32 psapi)
