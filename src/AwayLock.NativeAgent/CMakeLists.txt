cmake_minimum_required(VERSION 3.20)
project(AwayLockNative LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(AwayLock.Agent WIN32 main.cpp agent.rc)
target_compile_definitions(AwayLock.Agent PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AwayLock.Agent PRIVATE /W4 /EHsc /permissive- /utf-8 /O1 /Gy /guard:cf)
target_link_options(AwayLock.Agent PRIVATE /OPT:REF /OPT:ICF /guard:cf)
target_link_libraries(AwayLock.Agent PRIVATE windowsapp wtsapi32 advapi32 shell32 user32)
add_executable(AwayLock.NativePolicyTests ../../tests/AwayLock.NativePolicyTests/main.cpp)
target_include_directories(AwayLock.NativePolicyTests PRIVATE .)
target_compile_options(AwayLock.NativePolicyTests PRIVATE /W4 /EHsc /utf-8)
add_executable(AwayLock.NativeBluetoothProbe ../../tests/AwayLock.NativeBluetoothProbe/main.cpp)
target_compile_definitions(AwayLock.NativeBluetoothProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AwayLock.NativeBluetoothProbe PRIVATE /W4 /EHsc /permissive- /utf-8 /O1 /Gy)
target_link_options(AwayLock.NativeBluetoothProbe PRIVATE /OPT:REF /OPT:ICF)
target_link_libraries(AwayLock.NativeBluetoothProbe PRIVATE windowsapp wtsapi32 advapi32 shell32 user32 psapi)
